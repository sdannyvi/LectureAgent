\section{Part 1: Understanding the Paper}

% -------------------------------------------------------
\begin{frame}{Story: a model makes a treatment recommendation}

A hospital deploys a neural network to recommend cancer treatment.
\bigskip

\begin{itemize}
  \item A new patient's profile --- age, genetics, history --- is unlike
        anyone in the training data.
  \item The model has seen no one like this under either treatment arm.
  \item The model outputs: \textbf{``Recommend treatment.\ Confidence: 94\%.''}
\end{itemize}

\bigskip
\begin{alertblock}{The problem}
Standard causal neural networks have no mechanism to say ``I don't know.''
They extrapolate silently and confidently into uncharted covariate space.
\end{alertblock}
\end{frame}

% -------------------------------------------------------
\begin{frame}{What is a causal neural network?}

\begin{columns}
\column{0.52\textwidth}
\textbf{Standard ML:} learn $f(x) \approx \E[Y \mid X=x]$.\\[8pt]

\textbf{Causal ML:} learn \emph{two} functions simultaneously ---
one per treatment arm:
\[
\mu_0(x) = \E[Y \mid X=x,\, T=0]
\]
\[
\mu_1(x) = \E[Y \mid X=x,\, T=1]
\]
\[
\hat\tau(x) = \mu_1(x) - \mu_0(x)
\]

\textbf{Key constraint:} for each unit we only ever observe
\emph{one} arm ($T\cdot Y(1) + (1-T)\cdot Y(0)$).
The other is the missing counterfactual.

\column{0.44\textwidth}
\begin{center}
\begin{tikzpicture}[node distance=0.9cm, every node/.style={font=\small}]
  \node[obs] (x)   at (0, 0)    {$X$};
  \node[obs] (phi) at (0, 1.6)  {$\phi(X)$};
  \node[above=0.2cm of phi, font=\tiny] {shared repr.};
  \node[obs] (m0)  at (-1.8, 3.2) {$\hat\mu_0$};
  \node[obs] (m1)  at ( 1.8, 3.2) {$\hat\mu_1$};
  \node[note] (tau) at (0, 4.8) {$\hat\tau(x)=\hat\mu_1-\hat\mu_0$};
  \draw[arrow] (x) -- (phi);
  \draw[arrow] (phi) -- (m0);
  \draw[arrow] (phi) -- (m1);
  \draw[arrow] (m0.north) -- (tau.south west);
  \draw[arrow] (m1.north) -- (tau.south east);
  \node[font=\tiny, below=0.05cm of m0] {$T\!=\!0$ head};
  \node[font=\tiny, below=0.05cm of m1] {$T\!=\!1$ head};
\end{tikzpicture}
\end{center}
\end{columns}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Course recap: CATE and the identification chain}

Individual-level causal estimand:
\[
\tau(x) = \E\bigl[Y(1)-Y(0) \mid X=x\bigr]
\]

Under \textbf{unconfoundedness} $(Y(1),Y(0)) \indep T \mid X$
and \textbf{consistency}:
\[
\tau(x)
= \underbrace{\E[Y \mid T=1,\, X=x]}_{\mu_1(x)}
- \underbrace{\E[Y \mid T=0,\, X=x]}_{\mu_0(x)}
\]

\begin{itemize}
  \item $\mu_1(x)$ is estimated from \emph{treated} training units near $x$.
  \item $\mu_0(x)$ is estimated from \emph{untreated} training units near $x$.
  \item Both estimates require \textbf{positivity}: $\;0 < e(x) < 1$.
\end{itemize}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Positivity fails silently in high dimensions}

\begin{block}{Positivity assumption}
\[
0 < \Prob(T=1 \mid X=x) < 1
\quad \text{for all } x \text{ in the support of } X.
\]
\end{block}

\bigskip
\begin{itemize}
  \item In low dimensions: checked with propensity score histograms.
  \item In high dimensions (images, EHRs, genomics): most patients are
        unique --- positivity fails somewhere for almost everyone.
  \item Consequence: some $\mu_1(x)$ or $\mu_0(x)$ must be
        \textbf{extrapolated}, not estimated from nearby data.
  \item \alert{Standard causal neural networks extrapolate silently,
        with no signal to the user.}
\end{itemize}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Classical fix: propensity score trimming}

\begin{enumerate}
  \item Estimate $\hat{e}(x) = \hat{\Prob}(T=1 \mid X=x)$.
  \item \textbf{Trim:} remove units where $\hat{e}(x) < \eta$ or
        $\hat{e}(x) > 1-\eta$.
  \item Estimate CATE only on the remaining data.
\end{enumerate}

\bigskip
\textbf{Intuition:} units near $e(x)=0$ have almost no counterparts in the
$T=1$ arm; units near $e(x)=1$ have almost no counterparts in the $T=0$ arm.
Both make $\mu_1(x)$ or $\mu_0(x)$ unreliable.

\bigskip
\begin{block}{The fundamental limit}
The propensity $e(x)$ captures \emph{marginal} treatment probability.
It does not directly tell you \emph{which arm} lacks data for a given $x$,
nor by how much.
\end{block}
\end{frame}

% -------------------------------------------------------
\begin{frame}{CEMNIST: a synthetic benchmark on real images}

\textbf{CEMNIST = Causal Effect MNIST} --- introduced by the authors as a new benchmark.

\bigskip
\begin{columns}
\column{0.55\textwidth}
\begin{itemize}
  \item \textbf{Covariates $X$:} real MNIST handwritten digit images
        (28$\times$28 pixels, digits 0--9).
  \item \textbf{Treatment $T$:} binary, synthetically assigned ---
        think of it as ``received a new medication.''
  \item \textbf{Outcome $Y$:} binary, synthetically defined ---
        think of it as ``recovered.''
  \item \textbf{Why synthetic?} So the true CATE is \emph{known by construction}
        and we can evaluate estimator quality.
\end{itemize}

\column{0.42\textwidth}
\begin{alertblock}{This is a toy model}
Treatment, outcome, and CATE values are \emph{not real}.\\
The point is to create a controlled, high-dimensional setting where we can stress-test causal methods.
\end{alertblock}
\end{columns}
\end{frame}

% -------------------------------------------------------
\begin{frame}{CEMNIST: data generating process}

Treatment and outcome probabilities are defined \emph{per digit class}:

\begin{center}
\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{lccccc}
\toprule
Digit(s) & $p(x)$ & $e(x)=\Prob(T{=}1\mid x)$ & $\Prob(Y{=}1\mid x,T{=}0)$ & $\Prob(Y{=}1\mid x,T{=}1)$ & CATE \\
\midrule
9          & 0.50  & $1/9 \approx 0.11$ & 1 & 0 & $-1$ \\
2          & 0.056 & $1.00$             & 0 & 1 & $+1$ \\
other odd  & 0.056 & $0.50$             & 1 & 0 & $-1$ \\
other even & 0.056 & $0.50$             & 0 & 1 & $+1$ \\
\bottomrule
\end{tabular}
\end{center}

\bigskip
\begin{columns}
\column{0.48\textwidth}
\textbf{Digit 9} (50\% of data): mostly untreated ($e=1/9$), but
many \emph{untreated} nines exist in training.\\
$\Rightarrow$ we can estimate \textbf{both} $\mu_0(9)$ and $\mu_1(9)$.\\
CATE is \textbf{estimable}.

\column{0.48\textwidth}
\textbf{Digit 2} (5.6\% of data): \emph{always} treated ($e=1$).\\
No untreated twos exist in training.\\
$\Rightarrow$ $\mu_0(2)$ must be \textbf{extrapolated}.\\
CATE is \textbf{not estimable}.
\end{columns}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Why propensity trimming fails on CEMNIST}

\begin{columns}
\column{0.52\textwidth}

A trimmer sets thresholds $\eta$ (lower) and $1-\eta$ (upper):

\medskip
\begin{itemize}
  \item \textbf{Upper threshold} (e.g.\ $1-\eta = 0.75$): removes units with
        $\hat e(x) > 0.75$.
        Digit 2 has $e=1$ $\Rightarrow$ correctly trimmed. \textcolor{green!60!black}{\checkmark}
  \medskip
  \item \textbf{Lower threshold} (e.g.\ $\eta = 0.05$--$0.10$): removes units
        with $\hat e(x) < \eta$.
        Digit 9 has $e=1/9\approx 0.11$, and an estimated propensity model
        often pushes nines even closer to 0 ---
        \alert{nines get trimmed too.} $\boldsymbol{\times}$
\end{itemize}

\medskip
\textbf{Result:} the trimmer removes both twos (correct) \emph{and} nines
(incorrect --- nines have plenty of data on both arms).

\column{0.46\textwidth}
\includegraphics[width=\textwidth,height=0.62\textheight,keepaspectratio]{fig2a_cemnist_propensity.png}
\\\centering{\footnotesize Fig.\ 2a: estimated propensity histogram.
Large spike at left = untreated nines (many, low propensity).
Trimming near $\eta\approx0.10$ removes nines, not just twos.}
\end{columns}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Core insight: non-overlap is covariate shift}

Each arm model is trained on a \emph{different} slice of the data:
$\mu_0$ on $p(X\mid T=0)$, \; $\mu_1$ on $p(X\mid T=1)$.
At test time we evaluate \emph{both} at a new $x$.

\medskip
\begin{block}{Key observation}
$e(x)\approx 0$: almost no treated unit like $x$ in training
$\;\Rightarrow\;$ $x$ is OoD for $\mu_1$.\\[3pt]
$e(x)\approx 1$: almost no untreated unit like $x$ in training
$\;\Rightarrow\;$ $x$ is OoD for $\mu_0$.
\end{block}

\medskip
\begin{columns}
\column{0.36\textwidth}
\begin{center}
\begin{tikzpicture}
  \node[conf]  (X) at ( 0,  1.5) {$X$};
  \node[treat] (T) at (-1.6, 0)  {$T$};
  \node[outc]  (Y) at ( 1.6, 0)  {$Y$};
  \draw[arrow] (X) -- (T);
  \draw[arrow] (X) -- (Y);
  \draw[arrow] (T) -- (Y);
\end{tikzpicture}
\end{center}
\column{0.60\textwidth}
\small
\begin{block}{Non-overlap $=$ covariate shift}
If overlap fails at $x$, at least one arm model is asked to predict outside its training support --- the classical covariate shift problem.\\[3pt]
This opens the door to OoD detection tools from ML.
\end{block}
\end{columns}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Why does Bayes connect propensity to OoD?}

We want to know: ``Is $x$ OoD for the $T=1$ model?''\\
$\Leftrightarrow$ Is $p(x \mid T=1)$ small?

\bigskip
\begin{block}{Bayes' theorem}
\[
p(x \mid T=1)
= \frac{p(T=1 \mid x)\cdot p(x)}{p(T=1)}
= \frac{e(x)\cdot p(x)}{p(T=1)}
\]
\end{block}

\medskip
\begin{itemize}
  \item $p(T=1)$ and $p(x)$ are fixed population quantities, so
        $p(x \mid T=1) \propto e(x)$.
  \item $e(x)\approx 0 \;\Rightarrow\; p(x\mid T=1)\approx 0 \;\Rightarrow\; x$ OoD for $\mu_1$.
  \item $e(x)\approx 1 \;\Rightarrow\; p(x\mid T=0)\approx 0 \;\Rightarrow\; x$ OoD for $\mu_0$.
\end{itemize}

\smallskip
\begin{block}{Why this matters}
$p(x\mid T)$ is not directly observable, but $e(x)=p(T=1\mid x)$ \emph{is} estimable.
Bayes makes OoD-ness \textbf{operationalizable} via the propensity score.
\end{block}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Epistemic vs.\ aleatoric uncertainty}

\begin{columns}
\column{0.48\textwidth}
\begin{block}{Aleatoric uncertainty}
Inherent noise in the outcome $Y$.\\[4pt]
\textit{Example:} even knowing the true $\mu_t(x)$ exactly, there is still
randomness in $Y$.\\[4pt]
\textbf{Cannot} be reduced with more training data.
\end{block}

\vspace{8pt}

\begin{block}{Epistemic uncertainty}
Uncertainty from \emph{lack of training data} at $x$.\\[4pt]
\textit{Example:} $x$ is OoD for $\mu_1$ (digit 2 with $e=1$) --- many possible
functions $\mu_1$ fit the training data equally well.\\[4pt]
\textbf{Disappears} as more relevant data arrives.
\end{block}

\column{0.48\textwidth}
\begin{center}
We want to detect\\[6pt]
\textbf{epistemic} uncertainty\\[6pt]
in $\hat\tau(x) = \hat\mu_1(x)-\hat\mu_0(x)$,\\[6pt]
and use it to decide\\
when to withhold a recommendation.
\end{center}
\end{columns}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Bayesian Neural Networks (BNNs): the tool}

Replace a single fixed network $\mu_0(x;\,\omega_0)$ with a
\emph{distribution over networks}:
\[
p(\omega_0 \mid \mathcal{D}) \quad \text{(posterior over weights given training data).}
\]

Epistemic uncertainty $=$ variance of predictions across BNN draws:
\[
U_0(x) = \mathrm{Var}_{\omega_0 \sim p(\omega_0\mid\mathcal{D})}
          \bigl[\mu_0(x;\,\omega_0)\bigr].
\]

\begin{itemize}
  \item \textbf{In-distribution $x$:} all weight draws $\omega_0$ give similar
        predictions $\Rightarrow$ low variance.
  \item \textbf{OoD $x$:} different $\omega_0$ give wildly different predictions
        $\Rightarrow$ high variance.
\end{itemize}

\bigskip
\begin{block}{Practical approximation: MC Dropout (Gal \& Ghahramani, 2016)}
Add dropout to each outcome head; keep dropout \emph{on} at test time.\\
Each stochastic forward pass $\approx$ one sample from $p(\omega\mid\mathcal{D})$.\\
Run $S$ passes; estimate $U(x)$ as the sample variance. Cheap.
\end{block}
\end{frame}

\begin{frame}{CATE uncertainty: step-by-step derivation}

For a single BNN draw $\omega = (\omega_0, \omega_1)$, define
$\hat\tau_\omega(x) = \mu_{\omega_1}(x) - \mu_{\omega_0}(x)$.

Apply the \textbf{law of total variance} conditioning on $\omega$:
\begin{align*}
\mathrm{Var}_{Y,\omega}[Y(1){-}Y(0)\mid x]
&= \underbrace{\E_\omega\!\bigl[\mathrm{Var}_Y[Y(1){-}Y(0)\mid \mu_{\omega_1}(x),\mu_{\omega_0}(x)]\bigr]}_{\text{aleatoric: noise in }Y\text{ even if we knew }\omega}\\[4pt]
&\;+\; \underbrace{\mathrm{Var}_\omega\!\bigl[\E_Y[Y(1){-}Y(0)\mid \mu_{\omega_1}(x),\mu_{\omega_0}(x)]\bigr]}_{\text{epistemic}}
\end{align*}

The epistemic term simplifies because
$\E_Y[Y(t)\mid \mu_{\omega_t}(x)] = \mu_{\omega_t}(x)$:
\[
\text{epistemic} = \mathrm{Var}_\omega[\mu_{\omega_1}(x)-\mu_{\omega_0}(x)]
= \mathrm{Var}_\omega[\hat\tau_\omega(x)] =: U(x).
\]

\begin{block}{When is $U(x)$ large?}
When different $\omega$ draws give very different $\hat\tau_\omega(x)$ ---
exactly when $\mu_{\omega_1}$ or $\mu_{\omega_0}$ is OoD at $x$.
\end{block}
\end{frame}

\begin{center}

\midrule

\bigskip
\textbf{Paper's recipe:} add MC Dropout to the outcome heads of any model
$\Rightarrow$ get $U(x)$ for free, with minimal code changes.
\end{frame}

% -------------------------------------------------------
\begin{frame}{Rejection policy: thresholding on uncertainty}

\textbf{Decision rule:} given rejection budget $r_{\mathrm{rej}}$,
withhold recommendation for the $r_{\mathrm{rej}}$ fraction with highest $U(x)$.

\bigskip
\begin{columns}
\column{0.46\textwidth}
\textbf{Uncertainty policy}
\[
\text{reject } x \;\text{ if }\; U(x) > \kappa
\]
Detects non-overlap and covariate shift.

\column{0.46\textwidth}
\textbf{Propensity trimming}
\[
\text{reject } x \;\text{ if }\; \hat{e}(x) < \eta \;\text{ or }\; > 1{-}\eta
\]
Marginal probability only.\\
Trims nines, keeps twos (CEMNIST).
\end{columns}

\bigskip
\textbf{Metric:} recommendation error rate and $\sqrt{\mathrm{PEHE}}$
on the non-rejected subset, at fixed $r_{\mathrm{rej}}$.
\end{frame}

\begin{frame}{Part 1 recap}

\begin{block}{What Jesson et al.\ (2020) show}
\begin{enumerate}
  \item Non-overlap $=$ covariate shift for one outcome model.
  \item Epistemic uncertainty in BNNs detects this --- and detects
        input covariate shift as a bonus.
  \item MC Dropout makes this cheap and practical.
  \item Uncertainty-based rejection outperforms propensity trimming,
        especially when $e(x)$ and CATE estimability are decorrelated.
\end{enumerate}
\end{block}

\begin{center}
\Large\textit{Now: how much should we trust this?}
\end{center}
\end{frame}

\begin{frame}{Assumption audit}

\begin{center}

\small
\begin{tabular}{p{5.5cm}p{2.8cm}p{3.5cm}}

  & \alert{Assumed, untested}

\bigskip
\begin{alertblock}{}
The paper relaxes positivity. It inherits all other classical assumptions intact.
\end{alertblock}

% -------------------------------------------------------
\begin{frame}{The unconfoundedness blindspot}

The method tells you \textbf{when the model is uncertain}.
It does not tell you \textbf{whether the model is biased}.

\bigskip
If unconfoundedness fails --- a hidden $U$ confounds both $T$ and $Y$:

\begin{center}
\begin{tikzpicture}[node distance=2cm]
  \node[treat] (T) {$T$};
  \node[outc, right=of T] (Y) {$Y$};
  \node[conf, above=1.2cm of T] (X) {$X$};
  \node[hidden, above right=0.8cm and 0.5cm of T] (U) {$U$};
  \draw[arrow] (T) -- (Y);
  \draw[arrow] (X) -- (T);
  \draw[arrow] (X) -- (Y);
  \draw[badpath] (U) -- (T);
  \draw[badpath] (U) -- (Y);
\end{tikzpicture}
\end{center}

\begin{itemize}
  \item $\hat\mu_1(x)$ and $\hat\mu_0(x)$ both absorb the confounding signal.
  \item \alert{$U(x) = \mathrm{Var}_\omega[\hat\tau(x)]$ can be small even when bias is large.}
\end{itemize}

\begin{block}{}
A Bayesian causal model can be \emph{confidently wrong} under hidden confounding.
\end{block}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Two failure modes: only one is addressed}

\begin{center}
\renewcommand{\arraystretch}{1.8}
\begin{tabular}{r|p{4.6cm}p{4.6cm}}
 & \textbf{Low bias} & \textbf{High bias (hidden confounding)} \\
\hline
\textbf{High $U(x)$}
  & \textcolor{green!60!black}{Reject $\Rightarrow$ safe.}
  & \textcolor{green!60!black}{Reject $\Rightarrow$ safe (by luck).} \\
\hline
\textbf{Low $U(x)$}
  & \textcolor{green!60!black}{Recommend $\Rightarrow$ safe.}
  & \alert{Recommend $\Rightarrow$ confidently wrong.}
    \textbf{Blind spot.} \\
\end{tabular}
\end{center}

\bigskip
The paper operates in the left column and top row.\\
The bottom-right cell is untouched.
\end{frame}

% -------------------------------------------------------
\begin{frame}{What about sensitivity analysis?}

Classical tools address hidden confounding directly:

\begin{itemize}
  \item \textbf{Rosenbaum bounds:} how large must hidden $U$ be to
        overturn the conclusion?
  \item \textbf{E-values:} minimum confounder strength to explain away the effect.
  \item \textbf{Random common cause refuters} (DoWhy): add a synthetic
        unobserved confounder; check if the estimate changes.
\end{itemize}

\bigskip
\begin{alertblock}{The paper has no sensitivity analysis.}
It tells us \emph{when} to trust a CATE estimate (overlap, in-distribution).
It does not tell us \emph{how robust} the estimate is to hidden confounding.
\end{alertblock}

\bigskip
Uncertainty quantification and sensitivity analysis are \textbf{orthogonal}.
A complete pipeline needs both.
\end{frame}

% -------------------------------------------------------
\begin{frame}{The benchmarking problem}

All experiments use \textbf{semi-synthetic} datasets:

\begin{itemize}
  \item \textbf{IHDP}: real covariates from a clinical trial; synthetic outcomes.
  \item \textbf{ACIC 2016}: synthetic competition data; known ground truth.
  \item \textbf{CEMNIST}: MNIST digits; synthetic treatment and outcomes.
\end{itemize}

\bigskip
In semi-synthetic data, true CATE is known by construction
$\Rightarrow$ $\sqrt{\mathrm{PEHE}}$ is computable.

\bigskip
\begin{alertblock}{In real-world observational data}
$Y_i(0)$ for treated units is \textbf{never observed}.\\
$\sqrt{\mathrm{PEHE}}$ is \textbf{uncomputable}.\\
The paper's main metric cannot be verified in deployment.
\end{alertblock}
\end{frame}

\begin{frame}{What the paper genuinely contributes}

\begin{enumerate}
  \item \textbf{Elegant reframing.}
        Non-overlap $=$ covariate shift for one arm.
        Opens a toolkit (BNNs, OoD detection) not previously applied here.

  \item \textbf{Practical method.}
        MC Dropout adds uncertainty to any neural causal model
        with minimal engineering cost.

  \item \textbf{New benchmark.}
        CEMNIST exposes a concrete failure mode of propensity trimming
        in high-dimensional, reproducible settings.

  \item \textbf{Better rejection.}
        Outperforms propensity trimming on IHDP, ACIC, and CEMNIST
        across five model architectures.
\end{enumerate}

\bigskip
\begin{block}{What it does not contribute}
A solution to hidden confounding, a sensitivity analysis, or a way to
validate CATE estimates where ground truth is unavailable.
\end{block}
\end{frame}

\begin{frame}{Discussion questions --- Part A}

\begin{enumerate}

  \item \textbf{Uncertainty and confounding.}
        The paper assumes unconfoundedness throughout.
        If there are hidden confounders, can high $U(x)$ protect you?
        What would need to change to also catch confounding bias?

  \bigskip
  \item \textbf{Real-world validation.}
        The results are on semi-synthetic data.
        What experiment --- using only observational data without known CATE ---
        would convince you this holds in clinical practice?

  \bigskip
  \item \textbf{Redundancy with CFRNet.}
        CFRNet already penalises for overlap via MMD regularisation.
        Does MC Dropout on top give something genuinely new,
        or is it partially redundant?

\end{enumerate}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Discussion questions --- Part B}

\begin{enumerate}
  \setcounter{enumi}{3}

  \item \textbf{What has the model contributed?}
        You run the method and get high $U(x)$ for a patient.
        You withhold the recommendation and defer to a clinician.
        What has the causal model actually contributed?
        Is detecting uncertainty enough?

  \bigskip
  \item \textbf{Uncertainty about what, exactly?}
        The paper estimates $U(x) = \mathrm{Var}_\omega[\hat\tau(x)]$.
        Is this the same as uncertainty about the true causal effect $Y(1)-Y(0)$?
        What components of total uncertainty are missing?

\end{enumerate}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Take-away}

\[
U(x) = \mathrm{Var}_\omega\bigl[\hat\mu_{\omega_1}(x) - \hat\mu_{\omega_0}(x)\bigr]
\]

\begin{itemize}
  \item \textbf{What the paper gives you:}
        a principled signal for when to withhold a CATE estimate ---
        detecting non-overlap and covariate shift simultaneously.

  \item \textbf{What it assumes and does not question:}
        unconfoundedness, correct model class, i.i.d.\ data.
        High $U(x)$ is not a guard against confounding bias.
\end{itemize}

\bigskip
\begin{block}{The deeper lesson}
\textbf{Identification and uncertainty quantification are orthogonal.}\\
You can know exactly how uncertain you are about the wrong estimand.\\
Epistemic uncertainty $\neq$ robustness to assumption violations.
\end{block}
\end{frame}
