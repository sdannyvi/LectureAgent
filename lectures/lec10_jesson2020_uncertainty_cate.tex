% =======================================================
\section{Part 1: Understanding the Paper}
% =======================================================

% -------------------------------------------------------
\begin{frame}{Story: a model makes a treatment recommendation}

A hospital deploys a neural network to recommend cancer treatment.
\bigskip

\begin{itemize}
  \item A new patient's profile --- age, genetics, history --- is unlike
        anyone in the training data.
  \item The model has seen no one like this under either treatment arm.
  \item The model outputs: \textbf{``Recommend treatment.\ Confidence: 94\%.''}
\end{itemize}

\bigskip
\begin{alertblock}{The problem}
Standard causal neural networks have no mechanism to say ``I don't know.''
They extrapolate silently and confidently into uncharted covariate space.
\end{alertblock}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Course recap: CATE and identification}

The individual-level estimand (from our notation):
\[
\tau(x) = \E\bigl[Y(1)-Y(0) \mid X=x\bigr]
\]

Under \textbf{unconfoundedness} $(Y(1),Y(0)) \indep T \mid X$
and \textbf{consistency}:
\[
\tau(x)
= \underbrace{\E[Y \mid T=1,\, X=x]}_{\mu_1(x)}
- \underbrace{\E[Y \mid T=0,\, X=x]}_{\mu_0(x)}
\]

\begin{itemize}
  \item $\mu_1(x)$ requires observed \emph{treated} units near $x$.
  \item $\mu_0(x)$ requires observed \emph{untreated} units near $x$.
  \item Both require \textbf{positivity}: $\;0 < e(x) < 1$.
\end{itemize}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Positivity fails silently in high dimensions}

\begin{block}{Positivity assumption}
\[
0 < \Prob(T=1 \mid X=x) < 1
\quad \text{for all } x \text{ in the support of } X.
\]
\end{block}

\bigskip
\begin{itemize}
  \item In low dimensions: checked with propensity score histograms.
  \item In high dimensions (images, EHRs, genomics): every individual is
        almost unique --- positivity fails somewhere for almost everyone.
  \item Consequence: some $\mu_1(x)$ or $\mu_0(x)$ must be
        \textbf{extrapolated}, not estimated.
  \item \alert{Standard causal neural networks do not flag when they are extrapolating.}
\end{itemize}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Classical fix: propensity score trimming}

\begin{enumerate}
  \item Estimate $\hat{e}(x) = \hat{\Prob}(T=1 \mid X=x)$.
  \item \textbf{Trim:} remove units where $\hat{e}(x) < \eta$ or
        $\hat{e}(x) > 1-\eta$.
  \item Estimate CATE on the remaining data.
\end{enumerate}

\bigskip
\textbf{Intuition:} units near $e(x)=0$ or $e(x)=1$ have few counterparts
in the opposite arm --- their CATE estimates are unreliable.

\bigskip
\begin{block}{The fundamental limit of trimming}
The propensity score captures \emph{marginal} probability $\Prob(T=1 \mid x)$.
It does not directly tell you which arm lacks data for a specific $x$.
As we will see, this can lead to trimming exactly the wrong units.
\end{block}
\end{frame}

% -------------------------------------------------------
\begin{frame}{When trimming fails: the CEMNIST story}

New benchmark introduced in the paper: \textbf{causal-effect MNIST}.

\bigskip
\begin{center}
\small
\begin{tabular}{lcccc}
\toprule
Digit(s)   & $p(x)$ & $e(x)=\Prob(T{=}1\mid x)$ & CATE & Estimable? \\
\midrule
9          & 0.50   & $1/9 \approx 0.11$         & $-1$ & \textcolor{green!60!black}{Yes --- many untreated nines} \\
2          & 0.056  & $1.00$                     & $+1$ & \alert{No --- only treated twos} \\
other odd  & 0.056  & $0.50$                     & $-1$ & \textcolor{green!60!black}{Yes} \\
other even & 0.056  & $0.50$                     & $+1$ & \textcolor{green!60!black}{Yes} \\
\bottomrule
\end{tabular}
\end{center}

\bigskip
\begin{itemize}
  \item Digit 9: low propensity ($\approx 0.11$), but plenty of untreated nines
        $\Rightarrow$ CATE \emph{is} estimable.
  \item Digit 2: propensity $= 1$, no untreated twos
        $\Rightarrow$ CATE is \emph{not} estimable.
  \item A trimmer with $\eta = 0.05$ \alert{keeps twos, removes nines} ---
        exactly backwards.
\end{itemize}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Core insight: non-overlap is covariate shift}

\textbf{Recall:} covariate shift = the test distribution differs from the training
distribution.

\bigskip
Consider estimating $\mu_1(x) = \E[Y \mid T=1,\, X=x]$.
The model for the $T=1$ arm is trained on data from $p(X \mid T=1)$.

\bigskip
\begin{block}{The reframing (Jesson et al.)}
If $e(x) = \Prob(T=1 \mid x) \approx 0$, then by Bayes' theorem:
\[
p(x \mid T=1) \approx 0.
\]
So $x$ is \textbf{out-of-distribution} for the $T=1$ model --- a covariate shift.
\end{block}

\bigskip
\begin{itemize}
  \item $e(x)\approx 0$: $\mu_1(x)$ is OoD.
  \item $e(x)\approx 1$: $\mu_0(x)$ is OoD.
  \item Non-overlap $\Leftrightarrow$ at least one outcome model must extrapolate.
\end{itemize}

\bigskip
\textbf{Consequence:} we can address both non-overlap \emph{and} input covariate
shift with a single tool --- epistemic uncertainty.
\end{frame}

% -------------------------------------------------------
\begin{frame}{Epistemic vs.\ aleatoric uncertainty}

\begin{columns}
\column{0.46\textwidth}
\begin{block}{Aleatoric uncertainty}
Inherent noise in $Y$.\\[4pt]
\textit{Example:} a fair coin --- even with unlimited data you cannot predict the flip.\\[4pt]
Cannot be reduced with more observations.
\end{block}

\column{0.46\textwidth}
\begin{block}{Epistemic uncertainty}
Uncertainty due to \emph{lack of data}.\\[4pt]
\textit{Example:} no treated units like this patient $\Rightarrow$ uncertain what $\mu_1(x)$ is.\\[4pt]
Disappears as more relevant data arrives.
\end{block}
\end{columns}

\bigskip
\begin{center}
We want to detect \textbf{epistemic} uncertainty in
$\hat{\tau}(x) = \hat{\mu}_1(x) - \hat{\mu}_0(x)$,\\
and use it to decide when to withhold a recommendation.
\end{center}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Bayesian neural networks: the tool}

Replace a single network $\mu_0(x;\,\omega_0)$ with a
\emph{distribution over networks}:
\[
p(\omega_0 \mid \mathcal{D}) \quad \text{(posterior over weights).}
\]

Epistemic uncertainty $\approx$ variance of predictions across sampled networks:
\[
U_0(x) = \mathrm{Var}_{\omega_0 \sim p(\omega_0\mid\mathcal{D})}
          \bigl[\mu_0(x;\,\omega_0)\bigr].
\]

\begin{itemize}
  \item \textbf{In-distribution $x$:} all networks agree $\Rightarrow$ low variance.
  \item \textbf{OoD $x$:} networks disagree $\Rightarrow$ high variance.
\end{itemize}

\bigskip
\begin{block}{Practical approximation: MC Dropout (Gal \& Ghahramani, 2016)}
Add dropout to each outcome head.
Keep dropout \emph{on} at test time.
Each stochastic forward pass $\approx$ one draw from the approximate posterior.
\end{block}
\end{frame}

% -------------------------------------------------------
\begin{frame}{CATE uncertainty: the formal decomposition}

By the law of total variance:
\[
\mathrm{Var}\bigl[Y(1)-Y(0) \mid x\bigr]
=
\underbrace{
  \E_\omega\!\bigl[
    \mathrm{Var}_{Y}\bigl[Y(1){-}Y(0)\mid
      \mu_{\omega_1}(x),\mu_{\omega_0}(x)\bigr]
  \bigr]
}_{\text{aleatoric}}
+
\underbrace{
  \mathrm{Var}_\omega\bigl[\mu_{\omega_1}(x) - \mu_{\omega_0}(x)\bigr]
}_{\text{epistemic in }\hat\tau(x)}
\]

\bigskip
The paper uses the epistemic term as its uncertainty signal:
\[
U(x) \;=\; \mathrm{Var}_\omega\bigl[\hat\tau(x)\bigr]
      \;=\; \mathrm{Var}_\omega\bigl[\mu_{\omega_1}(x) - \mu_{\omega_0}(x)\bigr].
\]

\begin{itemize}
  \item Estimated by sampling $\omega$ via MC Dropout at test time.
  \item $U(x)$ is large when at least one arm model is OoD at $x$.
\end{itemize}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Neural CATE estimators: a brief map}

\begin{center}
\small
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{lll}
\toprule
Model      & Key idea                                      & Bayesian extension \\
\midrule
T-Learner  & Fit $\mu_0$, $\mu_1$ separately               & B-TLearner \\
TARNet     & Shared representation, separate outcome heads & BTARNet \\
CFRNet     & TARNet + MMD regularisation for overlap       & BCFR-MMD \\
Dragonnet  & TARNet + joint propensity prediction head     & BDragonnet \\
CEVAE      & Latent confounders via VAE                    & BCEVAE + neg.\ sampling \\
\bottomrule
\end{tabular}
\end{center}

\bigskip
\textbf{Paper's recipe:} add MC Dropout to the outcome heads of any model
$\Rightarrow$ get $U(x)$ for free, with minimal code changes.

\bigskip
\textbf{CEVAE special case:} non-overlap inputs are generated by flipping
$t \to 1-t$ on training points (negative sampling) --- tractable because
the OoD distribution is known.
\end{frame}

% -------------------------------------------------------
\begin{frame}{Rejection policy: thresholding on uncertainty}

\textbf{Decision rule:} given a rejection budget $r_{\mathrm{rej}}$,
withhold recommendation for the $r_{\mathrm{rej}}$ fraction of test
units with highest $U(x)$.

\bigskip
\begin{columns}
\column{0.46\textwidth}
\textbf{Uncertainty policy}
\[
\text{reject } x \;\text{ if }\; U(x) > \kappa
\]
\begin{itemize}
  \item Model-based; detects non-overlap and covariate shift.
  \item Threshold $\kappa$ set to achieve target $r_{\mathrm{rej}}$.
\end{itemize}

\column{0.46\textwidth}
\textbf{Propensity trimming}
\[
\text{reject } x \;\text{ if }\; \hat{e}(x) < \eta \;\text{ or }\; > 1{-}\eta
\]
\begin{itemize}
  \item Only captures marginal probability.
  \item Trims low-$e$ units even when their CATE is estimable (digit 9).
\end{itemize}
\end{columns}

\bigskip
\textbf{Evaluation metric:} recommendation error rate and
$\sqrt{\mathrm{PEHE}}$ on non-rejected subset, at fixed $r_{\mathrm{rej}}$.
\end{frame}

\end{frame}

% -------------------------------------------------------
\begin{frame}{Part 1 recap}

\begin{block}{What Jesson et al.\ (2020) show}
\begin{enumerate}
  \item Non-overlap $=$ covariate shift for one outcome model.
  \item Epistemic uncertainty in BNNs detects this --- and detects
        input covariate shift as a bonus.
  \item MC Dropout is a cheap, practical approximation.
  \item Uncertainty-based rejection outperforms propensity trimming,
        especially when $e(x)$ and CATE estimability are decorrelated.
\end{enumerate}
\end{block}

\bigskip
\begin{center}
\Large\textit{Now: how much should we trust this?}
\end{center}
\end{frame}

\section{Part 2: Critical Evaluation}

% -------------------------------------------------------
\begin{frame}{Assumption audit}

\begin{center}
\renewcommand{\arraystretch}{1.5}
\small
\begin{tabular}{p{5.5cm}p{2.5cm}p{4cm}}
\toprule
Assumption & Paper status & Notes \\
\midrule
Unconfoundedness:\\
$(Y(1),Y(0))\indep T\mid X$
  & \alert{Assumed, untested}
  & Foundation of all CATE identification \\
SUTVA / consistency
  & \alert{Assumed, untested}
  & Standard in this literature \\
Positivity $0 < e(x) < 1$
  & \textcolor{green!60!black}{Relaxed}
  & The paper's contribution \\
Correct model class
  & \alert{Assumed}
  & NN may misspecify $\mu_t(x)$ \\
i.i.d.\ training data
  & \alert{Assumed}
  & Violated under deployment shift \\
\bottomrule
\end{tabular}
\end{center}

\bigskip
\begin{alertblock}{Bottom line}
The paper relaxes positivity. It inherits all other classical assumptions intact,
and tests none of them empirically.
\end{alertblock}
\end{frame}

% -------------------------------------------------------
\begin{frame}{The unconfoundedness blindspot}

The paper's method tells you \textbf{when the model is uncertain}.\\
It does not tell you \textbf{whether the model is biased}.

\bigskip
\[
\hat\tau(x) = \hat\mu_1(x) - \hat\mu_0(x)
\]

If unconfoundedness fails --- a hidden $U$ confounds both $T$ and $Y$:

\begin{center}
\begin{tikzpicture}[node distance=2cm]
  \node[treat] (T) {$T$};
  \node[out, right=of T] (Y) {$Y$};
  \node[conf, above=1.2cm of T] (X) {$X$};
  \node[hidden, above right=0.8cm and 0.5cm of T] (U) {$U$};
  \draw[arrow] (T) -- (Y);
  \draw[arrow] (X) -- (T);
  \draw[arrow] (X) -- (Y);
  \draw[badpath] (U) -- (T);
  \draw[badpath] (U) -- (Y);
\end{tikzpicture}
\end{center}

\begin{itemize}
  \item $\hat\mu_1(x)$ and $\hat\mu_0(x)$ absorb the confounding signal.
  \item Their difference $\hat\tau(x)$ is biased.
  \item \alert{$U(x) = \mathrm{Var}_\omega[\hat\tau(x)]$ can be small even when bias is large.}
\end{itemize}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Two failure modes: only one is addressed}

\begin{center}
\renewcommand{\arraystretch}{1.8}
\begin{tabular}{r|p{4.5cm}p{4.5cm}}
 & \textbf{Low bias} & \textbf{High bias (hidden confounding)} \\
\hline
\textbf{High $U(x)$}
  & \textcolor{green!60!black}{Reject $\Rightarrow$ safe.}
    Paper detects this.
  & \textcolor{green!60!black}{Reject $\Rightarrow$ safe (by luck).}
    Paper detects this too, but for the wrong reason. \\
\hline
\textbf{Low $U(x)$}
  & \textcolor{green!60!black}{Recommend $\Rightarrow$ safe.}
  & \alert{Recommend $\Rightarrow$ confidently wrong.}
    \textbf{The paper's blind spot.} \\
\end{tabular}
\end{center}

\bigskip
A Bayesian causal model can be \emph{confident and wrong} if the data is
confounded --- not just if overlap is violated.
\end{frame}

% -------------------------------------------------------
\begin{frame}{What about sensitivity analysis?}

Classical causal inference addresses hidden confounding directly:

\begin{itemize}
  \item \textbf{Rosenbaum bounds:} how large must a hidden $U$ be to
        overturn the conclusion?
  \item \textbf{E-values:} minimum association strength of a confounder
        to explain away the estimated effect.
  \item \textbf{Random common cause refuters} (DoWhy): add a synthetic
        unobserved confounder; check if the estimate changes.
\end{itemize}

\bigskip
\begin{alertblock}{The paper has no sensitivity analysis.}
It tells us \emph{when} to trust a CATE estimate (in-distribution, overlap region).
It does not tell us \emph{how robust} the estimate is to hidden confounding.
\end{alertblock}

\bigskip
These are \textbf{orthogonal questions}.
Uncertainty quantification and sensitivity analysis both belong in a complete
causal inference pipeline. This paper provides one, not both.
\end{frame}

% -------------------------------------------------------
\begin{frame}{The benchmarking problem}

All experiments use \textbf{semi-synthetic} datasets:

\begin{itemize}
  \item \textbf{IHDP}: real covariates from a clinical trial; synthetic outcome model.
  \item \textbf{ACIC 2016}: synthetic data from a methods competition; known DGP.
  \item \textbf{CEMNIST}: MNIST digits; synthetic treatment and outcome assignment.
\end{itemize}

\bigskip
In semi-synthetic data, the true CATE is \textbf{known by construction}
--- that is why $\sqrt{\mathrm{PEHE}}$ can be computed.

\bigskip
\begin{alertblock}{In real-world observational data}
$Y_i(0)$ for treated units is \textbf{never observed}.\\
$\sqrt{\mathrm{PEHE}}$ is \textbf{uncomputable}.\\
We cannot verify the paper's claims in deployment.
\end{alertblock}

\bigskip
This is not a flaw unique to this paper --- it is the fundamental evaluation
problem in all individual-level causal effect estimation.
\end{frame}

% -------------------------------------------------------
\begin{frame}{Comparison to classical methods}

\begin{center}
\small
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{p{3.5cm}p{4cm}p{4cm}}
\toprule
 & \textbf{Propensity trimming} & \textbf{Uncertainty rejection} \\
\midrule
Detects non-overlap
  & Yes, via $\hat e(x)$
  & Yes, more precisely \\
Handles covariate shift
  & No
  & Yes \\
Handles hidden confounding
  & No
  & No \\
Sensitivity analysis
  & Via Rosenbaum bounds / E-values
  & None \\
Computationally cheap
  & Yes
  & No --- requires BNN training \\
Principled threshold
  & Yes (Caliendo \& Kopeinig)
  & No principled rule given \\
Verifiable on real data
  & Yes (no CATE ground truth needed)
  & Yes, but performance unverifiable \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

% -------------------------------------------------------
\begin{frame}{What the paper genuinely contributes}

\begin{enumerate}
  \item \textbf{Elegant reframing.}
        Non-overlap = covariate shift for one arm.
        This is clean, theoretically well-motivated, and opens a new toolkit.

  \item \textbf{Practical method.}
        MC Dropout adds uncertainty to any neural causal model
        with minimal engineering cost.

  \item \textbf{New benchmark.}
        CEMNIST exposes a concrete, reproducible failure mode of
        propensity-based trimming in high-dimensional data.

  \item \textbf{Better rejection.}
        Outperforms propensity trimming consistently across three
        benchmarks and five model architectures.
\end{enumerate}

\bigskip
\begin{block}{What it does not contribute}
A solution to hidden confounding, a sensitivity analysis, or any mechanism
to validate CATE estimates on real-world data where ground truth is unavailable.
\end{block}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Discussion questions --- Part A}

\begin{enumerate}

  \item \textbf{Uncertainty and confounding.}
        The paper assumes unconfoundedness throughout.
        If there are hidden confounders, can high epistemic uncertainty $U(x)$
        protect you? What would need to change in the framework to also catch
        confounding bias, not just data sparsity?

  \bigskip
  \item \textbf{Real-world validation.}
        The paper shows uncertainty beats trimming on IHDP and CEMNIST --- both
        semi-synthetic.
        What experiment, using only observational data with no known ground truth,
        would convince you this result holds in a clinical deployment?

  \bigskip
  \item \textbf{Redundancy with CFRNet.}
        CFRNet already penalises for overlap violations via MMD regularisation.
        Does adding MC Dropout on top give something genuinely new,
        or is it partially redundant?
        What does each mechanism address, precisely?

\end{enumerate}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Discussion questions --- Part B}

\begin{enumerate}
  \setcounter{enumi}{3}

  \item \textbf{What has the model contributed?}
        Suppose you run the paper's method on a new patient and get
        very high $U(x)$.
        You withhold the recommendation and defer to a clinician.
        What has the causal model actually contributed to the decision?
        Is detecting uncertainty enough to be useful? What else would you need?

  \bigskip
  \item \textbf{Uncertainty about what, exactly?}
        The paper estimates $U(x) = \mathrm{Var}_\omega[\hat\tau(x)]$
        --- variance across BNN draws.
        Is this the same as uncertainty about the true causal effect
        $Y(1)-Y(0)$?
        What components of total uncertainty are missing from $U(x)$?
        (Think: model misspecification, hidden confounders, estimation noise.)

\end{enumerate}
\end{frame}

% -------------------------------------------------------
\begin{frame}{Final take-away}

\[
U(x) = \mathrm{Var}_\omega\bigl[\hat\mu_{\omega_1}(x) - \hat\mu_{\omega_0}(x)\bigr]
\]

\begin{itemize}
  \item \textbf{What the paper gives you:}
        a principled, model-based signal for when to withhold a CATE estimate,
        detecting non-overlap and covariate shift simultaneously.

  \item \textbf{What it assumes and does not question:}
        unconfoundedness, correct model class, i.i.d.\ data, and SUTVA.
        High $U(x)$ is not a guard against confounding bias ---
        a model can be uncertain for the right reason and still be wrong
        for a different reason.
\end{itemize}

\bigskip
\begin{block}{The deeper lesson for this course}
\textbf{Identification and uncertainty quantification are orthogonal.}\\
You can know exactly how uncertain you are about the wrong estimand.\\
Epistemic uncertainty $\neq$ robustness to assumption violations.
\end{block}
\end{frame}
